const Stack = createNativeStackNavigator();

// 2. Tela Principal: Lista de Países (FlatList)
function ListaPaisesScreen({ navigation }) {
  const renderItem = ({ item }) => (
    <TouchableOpacity
      style={styles.card}
      // Navega para 'Detalhes' passando o objeto inteiro do país via parâmetro
      onPress={() => navigation.navigate('Detalhes', { pais: item })}>
      <Text style={styles.bandeira}>{item.bandeira}</Text>
      <View style={styles.infoContainer}>
        <Text style={styles.nomePais}>{item.nome}</Text>
        <Text style={styles.subtitulo}>Capital: {item.capital}</Text>
      </View>
      <Text style={styles.seta}>›</Text>
    </TouchableOpacity>
  );

  return (
   
      <FlatList
        data={PAISES}
        keyExtractor={(item) => item.id}
        renderItem={renderItem}
        contentContainerStyle={styles.lista}
      />
  );
}

// 3. Tela Secundária: Detalhes do País
function DetalhesPaisScreen({ route }) {
  // Recupera o objeto 'pais' enviado pela tela anterior
  const { pais } = route.params;

  return (
    <View style={styles.detalhesContainer}>
      <View style={styles.headerDetalhes}>
        <Text style={styles.bandeiraGrande}>{pais.bandeira}</Text>
        <Text style={styles.tituloGrande}>{pais.nome}</Text>
        <Text style={styles.continente}>{pais.continente}</Text>
      </View>

      <View style={styles.boxInfo}>
        <InfoLinha label="Capital" valor={pais.capital} />
        <InfoLinha label="População" valor={pais.populacao} />
        <InfoLinha label="Território" valor={pais.territorio} />
        <InfoLinha label="Idioma Oficial" valor={pais.idioma} />
      </View>
    </View>
  );
}

// Componente auxiliar para organizar as linhas de informação
function InfoLinha({ label, valor }) {
  return (
    <View style={styles.linha}>
      <Text style={styles.label}>{label}:</Text>
      <Text style={styles.valor}>{valor}</Text>
    </View>
  );
}

// 4. Configuração principal da Navegação
export default function App() {
  return (
    <NavigationContainer>
      <Stack.Navigator initialRouteName="Lista">
        <Stack.Screen
          name="Lista"
          component={ListaPaisesScreen}
          options={{ title: 'Países do Mundo' }}
        />
        <Stack.Screen
          name="Detalhes"
          component={DetalhesPaisScreen}
          // Define o título do cabeçalho dinamicamente com o nome do país
          options={({ route }) => ({ title: route.params.pais.nome })}
        />
      </Stack.Navigator>
    </NavigationContainer>
  );
}
